##  1.3 โปรแกรมควบคุม (20 คะแนน)

```cpp
// === Motor Pins ===
#define IN1 13
#define IN2 12
#define ENA 11 // PWM
#define IN3 9
#define IN4 8
#define ENB 10 // PWM

// === Joystick Pins ===
#define JOY_X A0
#define JOY_Y A1

int speedA = 0;
int speedB = 0;
const int step = 26; // เพิ่ม/ลด ทีละ 10% ของ 255
unsigned long lastMove = 0;
const unsigned long debounceDelay = 300; // ms

void setup() {
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(ENA, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  pinMode(ENB, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  float x_volt = analogRead(JOY_X) * (5.0 / 1023.0);
  float y_volt = analogRead(JOY_Y) * (5.0 / 1023.0);
  unsigned long now = millis();

  // Dead zone
  bool center = (x_volt >= 2.0 && x_volt <= 3.0) && (y_volt >= 2.0 && y_volt <= 3.0);

  if (!center && now - lastMove > debounceDelay) {
    // ขวา -> เพิ่มความเร็ว Motor A
    if (x_volt >= 4.0 && y_volt >= 2.0 && y_volt <= 3.0) {
      speedA = min(speedA + step, 255);
    }

    // ซ้าย -> ลดความเร็ว Motor A
    else if (x_volt <= 1.0 && y_volt >= 2.0 && y_volt <= 3.0) {
      speedA = max(speedA - step, 0);
    }

    // ขึ้น -> เพิ่มความเร็ว Motor B
    else if (y_volt >= 4.0 && x_volt >= 2.0 && x_volt <= 3.0) {
      speedB = min(speedB + step, 255);
    }

    // ลง -> ลดความเร็ว Motor B
    else if (y_volt <= 1.0 && x_volt >= 2.0 && x_volt <= 3.0) {
      speedB = max(speedB - step, 0);
    }

    lastMove = now;
  }

  // ควบคุมมอเตอร์ A
  analogWrite(ENA, speedA);
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  // ควบคุมมอเตอร์ B
  analogWrite(ENB, speedB);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  // Debug
  Serial.print("Speed A: "); Serial.print(speedA);
  Serial.print(" | Speed B: "); Serial.println(speedB);

  delay(50);
}
```

## More :

**เพิ่มการใช้ปุ่มกด (SW) บนจอยสติ๊ก KY-023** เพื่อควบคุม **เปิด/ปิดการจ่ายพลังงานให้มอเตอร์ทั้งสองตัว** โดยใช้ขา `SW` เชื่อมต่อกับ Arduino

##  การเชื่อมต่อเพิ่ม:

| KY-023 Pin | Arduino UNO |
| --- | --- |
| SW | D7 |
- อย่าลืมใช้ **pull-up resistor** (หรือตั้งในโค้ด) เพราะปุ่มนี้จะต่อ GND เมื่อกด

##  เพิ่มในโค้ด (เวอร์ชันรวม + สวิตช์)

```cpp
// === Motor Pins ===
#define IN1 13
#define IN2 12
#define ENA 11 // PWM
#define IN3 9
#define IN4 8
#define ENB 10 // PWM

// === Joystick Pins ===
#define JOY_X A0
#define JOY_Y A1
#define JOY_SW 7

int speedA = 0;
int speedB = 0;
const int step = 26; // 10% ของ 255
bool motorPower = true; // เปิดมอเตอร์เริ่มต้น
unsigned long lastMove = 0;
const unsigned long debounceDelay = 300;
unsigned long lastSwitch = 0;

void setup() {
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(ENA, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  pinMode(ENB, OUTPUT);

  pinMode(JOY_SW, INPUT_PULLUP); // ใช้ internal pull-up

  Serial.begin(9600);
}

void loop() {
  float x_volt = analogRead(JOY_X) * (5.0 / 1023.0);
  float y_volt = analogRead(JOY_Y) * (5.0 / 1023.0);
  unsigned long now = millis();

  // เช็คปุ่มกด SW (เมื่อถูกกด จะต่อ GND => LOW)
  if (digitalRead(JOY_SW) == LOW && now - lastSwitch > 500) {
    motorPower = !motorPower;
    Serial.print("Motor Power Toggled: ");
    Serial.println(motorPower ? "ON" : "OFF");
    lastSwitch = now;
  }

  // Dead zone
  bool center = (x_volt >= 2.0 && x_volt <= 3.0) && (y_volt >= 2.0 && y_volt <= 3.0);

  if (!center && now - lastMove > debounceDelay) {
    if (x_volt >= 4.0 && y_volt >= 2.0 && y_volt <= 3.0) {
      speedA = min(speedA + step, 255);
    } else if (x_volt <= 1.0 && y_volt >= 2.0 && y_volt <= 3.0) {
      speedA = max(speedA - step, 0);
    } else if (y_volt >= 4.0 && x_volt >= 2.0 && x_volt <= 3.0) {
      speedB = min(speedB + step, 255);
    } else if (y_volt <= 1.0 && x_volt >= 2.0 && x_volt <= 3.0) {
      speedB = max(speedB - step, 0);
    }

    lastMove = now;
  }

  if (motorPower) {
    analogWrite(ENA, speedA);
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);

    analogWrite(ENB, speedB);
    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);
  } else {
    analogWrite(ENA, 0);
    analogWrite(ENB, 0);
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);
  }

  Serial.print("Speed A: "); Serial.print(speedA);
  Serial.print(" | Speed B: "); Serial.print(speedB);
  Serial.print(" | Motor: "); Serial.println(motorPower ? "ON" : "OFF");

  delay(50);
}
```

### ผลลัพธ์:

- กดปุ่มบน Joystick (SW) → สลับเปิด/ปิดมอเตอร์ (ON/OFF)
- LED ที่ L298N จะติด/ดับตามสถานะจ่ายไฟ
- มอเตอร์ไม่หมุนถ้า OFF (แม้ค่า speed ยังอยู่)
